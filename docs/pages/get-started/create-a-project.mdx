// ==========================================
// 4. شاشة الاختبار (Quiz Screen)
// ==========================================
function QuizScreen({ addXp }) {
  const [idx, setIdx] = useState(0);
  const [ans, setAns] = useState(null);
  const q = LingoData.quizzes[idx % LingoData.quizzes.length];

  const handleAns = (o) => {
    if (ans) return;
    setAns(o);
    if (o === q.correct) addXp(q.xp);
  };

  return (
    <View style={styles.screen}>
      <Text style={styles.progressText}>السؤال {idx + 1} من {LingoData.quizzes.length}</Text>
      <View style={styles.qCard}><Text style={styles.qText}>{q.q}</Text></View>
      {q.opts.map(o => (
        <TouchableOpacity 
          key={o} 
          style={[styles.optBtn, ans === o && {backgroundColor: o === q.correct ? '#4CAF50' : '#F44336'}]}
          onPress={() => handleAns(o)}
        >
          <Text style={[styles.optTxt, ans === o && {color: '#fff'}]}>{o}</Text>
        </TouchableOpacity>
      ))}
      {ans && (
        <TouchableOpacity style={styles.nextBtn} onPress={() => {setIdx(idx+1); setAns(null);}}>
          <Text style={styles.nextTxt}>السؤال التالي ➔</Text>
        </TouchableOpacity>
      )}
    </View>
  );
}

// ==========================================
// 5. شاشة التحدث (Speaking Screen)
// ==========================================
function SpeakingScreen({ addXp }) {
  const [active, setActive] = useState(false);
  const task = LingoData.speaking[0]; // نموذج

  return (
    <View style={[styles.screen, {alignItems: 'center'}]}>
      <Text style={styles.sectionTitle}>نادي المحادثة 🎙️</Text>
      <View style={styles.qCard}><Text style={styles.qText}>"{task.text}"</Text></View>
      <TouchableOpacity 
        style={[styles.micBtn, active && {backgroundColor: 'red'}]}
        onPressIn={() => setActive(true)}
        onPressOut={() => {setActive(false); addXp(20); Alert.alert("الذكاء الاصطناعي 🤖", "نطق ممتاز! لقد حصلت على 20 نقطة.");}}
      >
        <Text style={{fontSize: 40}}>{active ? "🛑" : "🎤"}</Text>
      </TouchableOpacity>
      <Text style={{marginTop: 20}}>{active ? "يتم الآن تحليل صوتك..." : "اضغط مطولاً للتحدث"}</Text>
    </View>
  );
}

// ==========================================
// 6. المحرك الرئيسي والتنقل (App Main)
// ==========================================
const Stack = createStackNavigator();

export default function App() {
  const [xp, setXp] = useState(0);
  const [isDark, setIsDark] = useState(false);

  useEffect(() => {
    AsyncStorage.getItem('user_xp').then(v => v && setXp(parseInt(v)));
  }, []);

  const addXp = (v) => {
    const newXp = xp + v;
    setXp(newXp);
    AsyncStorage.setItem('user_xp', newXp.toString());
  };

  return (
    <NavigationContainer>
      <Stack.Navigator screenOptions={{ headerStyle: {backgroundColor: '#7C4DFF'}, headerTintColor: '#fff' }}>
        <Stack.Screen name="Splash" component={SplashScreen} options={{headerShown: false}} />
        <Stack.Screen name="Home" options={{headerShown: false}}>
          {(p) => <HomeScreen {...p} xp={xp} streak={5} />}
        </Stack.Screen>
        <Stack.Screen name="Quiz">{(p) => <QuizScreen {...p} addXp={addXp} />}</Stack.Screen>
        <Stack.Screen name="Speaking">{(p) => <SpeakingScreen {...p} addXp={addXp} />}</Stack.Screen>
        <Stack.Screen name="Settings">
          {() => (
            <View style={[styles.screen, isDark && {backgroundColor: '#121212'}]}>
              <View style={styles.row}>
                <Text style={isDark && {color: '#fff'}}>الوضع الليلي</Text>
                <Switch value={isDark} onValueChange={setIsDark} />
              </View>
            </View>
          )}
        </Stack.Screen>
      </Stack.Navigator>
    </NavigationContainer>
  );
}
